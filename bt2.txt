pragma solidity ^0.8.0;

contract StudentDatabase {
    struct Student {
        string name;
        uint256 rollNumber;
        string studentClass;
    }

    Student[] private students;

    function addStudent(string memory name, uint256 rollNumber, string memory studentClass) public {
        students.push(Student(name, rollNumber, studentClass));
    }

    function getStudentByID(uint256 ID) public view returns (string memory, uint256, string memory) {
        require(ID < students.length, "Student does not exist in database");
        Student storage s = students[ID];
        return (s.name, s.rollNumber, s.studentClass);
    }

    function getTotalNumberOfStudents() public view returns (uint256) {
        return students.length;
        }
    }

//1. Aim of the program
To create a Student Database using a Solidity smart contract that can:
1. Add student details.
2. Retrieve student details using ID.
3. Display the total number of students.

5. Most important questions examiner may ask
Q1. Why did you use Solidity?

To create a smart contract for storing and retrieving student records on blockchain.

Q2. Why did you use an array?

To store multiple student records.

Q3. Why did you use a structure?

To store name, roll number and class together as one student record.

Q4. What is the difference between memory and storage?
- Memory: Temporary data.
- Storage: Permanent data stored on the blockchain.
Q5. Why is view used?

Because the function only reads data and does not change it.

Q6. Does adding a student require gas?

Yes. Calling addStudent() as a blockchain transaction modifies the contract's state, so it requires gas.

Q7. Does getting student details require gas?

A normal read-only call (view) does not require a gas payment from the caller. If executed as part of an on-chain transaction, it still consumes gas.
